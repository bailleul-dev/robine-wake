# Robine Wake Raspberry Pi Zero W boot configuration

kernel=zImage
disable_splash=1
gpu_mem=192

# Hardware buses and the I2S amplifier. Disable the PWM audio device so the
# amplifier receives a stable ALSA card identity.
dtparam=i2c_arm=on
dtparam=spi=on
dtparam=audio=off

# USB gadget (Ethernet over the data micro-USB port): first access to the board
# before Wi-Fi is configured (usb0, see config/target.exs).
dtoverlay=dwc2

# InnoMaker HiFi AMP Pro Mini HAT (Infineon MA12070P, I2S + I2C 0x20). It is a
# Merus-Audio design, driven by the ma120x0p codec as the ALSA card
# "snd_rpi_merus_amp". The driver accepts sample rates up to 48 kHz only.
# merus-amp-uart is the stock merus-amp overlay without its enable (GPIO14) and
# mute (GPIO15) lines, which are the UART pins to the ESP32 light controller
# (mini UART /dev/ttyS0). Use `dtoverlay=merus-amp` instead if the HAT needs
# those two lines and no ESP32 is wired. Ignore the HAT EEPROM, if any.
force_eeprom_read=0
dtoverlay=merus-amp-uart

# The camera feature is disabled; do not probe or load camera overlays.
camera_auto_detect=0

# The ramoops overlay works with the pstore driver to preserve crash
# information across reboots in DRAM.
dtoverlay=ramoops

# The onboard Bluetooth controller keeps the full-speed PL011 UART and is
# driven by the kernel (hci_uart, BlueZ) for the A2DP music receiver; the
# serial console stays on the mini UART.
enable_uart=1
dtparam=krnbt=on
core_freq=250

# The active LED is active low on the Zero. Heartbeat shows the CPU is alive.
dtparam=act_led_activelow=on
dtparam=act_led_trigger=heartbeat
