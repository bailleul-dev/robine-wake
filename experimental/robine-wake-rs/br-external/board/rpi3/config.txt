# Note: merus-amp (InnoMaker MA12070P Mini HAT) uses GPIO 14 (enable) and GPIO 15 (mute),
# so UART must remain disabled on [pi02] to avoid a pinmux conflict.
[pi3]
dtoverlay=hifiberry-dacplus
enable_uart=1
[pi02]
dtoverlay=merus-amp
enable_uart=0
[all]
dtparam=audio=off
dtoverlay=disable-bt
